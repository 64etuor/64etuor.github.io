const inBooksFolder = p => p.file.path.startsWith("200_Books/");
const pages = dv.pages("#src/book").where(inBooksFolder);

const done = pages.where(p => ["완료","done"].includes(p.status)).length;
const reading = pages.where(p => ["진행","active"].includes(p.status)).length;
const planned = pages.where(p => ["계획","inbox"].includes(p.status)).length;
const rated = pages.where(p => p.my_rate && p.my_rate > 0);
const avgRate = rated.length > 0
    ? (rated.map(p => p.my_rate).array().reduce((a,b) => a+b, 0) / rated.length).toFixed(1)
    : "-";
const totalPages = pages
    .map(p => p.total_page ?? 0).array()
    .reduce((a,b) => a+b, 0);

const container = document.createElement("div");
container.className = "stats-grid";

const stats = [
    { value: pages.length, label: "전체" },
    { value: done, label: "완료" },
    { value: reading, label: "진행 중" },
    { value: planned, label: "계획" },
    { value: avgRate, label: "평균 별점" },
    { value: totalPages.toLocaleString(), label: "총 페이지" },
];

stats.forEach(s => {
    const card = document.createElement("div");
    card.className = "stat-card";
    const val = document.createElement("div");
    val.className = "stat-value";
    val.textContent = s.value;
    const lbl = document.createElement("div");

const GOAL = 52;
const YEAR = new Date().getFullYear();
const monthNames = ["1월","2월","3월","4월","5월","6월","7월","8월","9월","10월","11월","12월"];

const pages = dv.pages("#src/book")
    .where(p => p.file.path.startsWith("200_Books/"))
    .where(p => ["완료","done"].includes(p.status))
    .where(p => {
        if (!p.finish_read_date) return false;
        const d = dv.date(p.finish_read_date);
        return d && d.year === YEAR;
    });

const done = pages.length;
const pct = Math.min(done / GOAL * 100, 100);

const monthlyCounts = new Array(12).fill(0);
pages.forEach(p => {
    const m = dv.date(p.finish_read_date).month - 1;
    monthlyCounts[m]++;
});

const now = new Date();
const curMonth = now.getMonth();

const endOfYear = new Date(YEAR, 11, 31);
const yearProgress = (now - startOfYear) / (endOfYear - startOfYear);
const expectedBooks = Math.round(GOAL * yearProgress);
const diff = done - expectedBooks;
const monthlyGoal = Math.ceil(GOAL / 12);
const thisMonthDone = monthlyCounts[curMonth];

    const goalLine = document.createElement("div");
    goalLine.className = "chart-goal-line";
    goalLine.style.bottom = `${(monthlyGoal / maxCount) * 100}%`;
    barWrapper.appendChild(goalLine);

    const bar = document.createElement("div");
    bar.className = "chart-bar";
    const h = maxCount > 0 ? (monthlyCounts[i] / maxCount) * 100 : 0;
    bar.style.height = `${h}%`;

    if (i === curMonth) bar.classList.add("current");
    else if (i > curMonth) bar.classList.add("future");
    if (monthlyCounts[i] >= monthlyGoal && monthlyCounts[i] > 0) bar.classList.add("achieved");

    if (monthlyCounts[i] > 0) {
        const countLabel = document.createElement("div");
        countLabel.className = "chart-count";
        countLabel.textContent = monthlyCounts[i];
        bar.appendChild(countLabel);
    }

    barWrapper.appendChild(bar);
    col.appendChild(barWrapper);

    const label = document.createElement("div");
    label.className = "chart-label";
    label.textContent = `${i + 1}`;
    col.appendChild(label);

    chartDiv.appendChild(col);
}

rightSection.appendChild(chartDiv);
wrapper.appendChild(rightSection);

dv.container.appendChild(wrapper);

독서 히트맵 — 완독일 + 독서 기간 표시

const YEAR = new Date().getFullYear();

const pages = dv.pages("#src/book")
    .where(p => p.file.path.startsWith("200_Books/"))
    .where(p => ["완료","done"].includes(p.status))
    .where(p => {
        if (!p.finish_read_date) return false;
        const d = dv.date(p.finish_read_date);
        return d && d.year === YEAR;
    });

const dateCounts = {};
pages.forEach(p => {
    const d = dv.date(p.finish_read_date).toFormat("yyyy-MM-dd");
    dateCounts[d] = (dateCounts[d] || 0) + 1;
});

const readingDays = {};
pages.forEach(p => {
    if (!p.start_read_date || !p.finish_read_date) return;
    const start = dv.date(p.start_read_date);
    const end = dv.date(p.finish_read_date);
    if (!start || !end) return;
    let cur = start;
    while (cur <= end) {
        const key = cur.toFormat("yyyy-MM-dd");
        readingDays[key] = (readingDays[key] || 0) + 1;
        cur = cur.plus({ days: 1 });
    }
});

const container = document.createElement("div");
container.className = "heatmap-container";

const months = ["1월","2월","3월","4월","5월","6월","7월","8월","9월","10월","11월","12월"];
const dayNames = ["일","월","화","수","목","금","토"];
const jan1 = new Date(YEAR, 0, 1);
const jan1Dow = jan1.getDay();
const today = new Date();

const monthStartWeek = {};
for (let mi = 0; mi < 12; mi++) {
    const first = new Date(YEAR, mi, 1);
    const dayOfYear = Math.round((first - jan1) / 86400000);
    monthStartWeek[Math.floor((dayOfYear + jan1Dow) / 7)] = mi;
}

const headerRow = document.createElement("div");
headerRow.className = "heatmap-header";
const emptyLabel = document.createElement("div");
emptyLabel.className = "heatmap-day-label";
headerRow.appendChild(emptyLabel);
for (let week = 0; week < 53; week++) {
    const slot = document.createElement("div");
    slot.className = "heatmap-month-label";
    if (monthStartWeek[week] !== undefined) {
        slot.textContent = months[monthStartWeek[week]];
    }
    headerRow.appendChild(slot);
}
container.appendChild(headerRow);

for (let dow = 0; dow < 7; dow++) {
    const row = document.createElement("div");
    row.className = "heatmap-row";

    const dayLabel = document.createElement("div");
    dayLabel.className = "heatmap-day-label";
    dayLabel.textContent = (dow === 1 || dow === 3 || dow === 5) ? dayNames[dow] : "";
    row.appendChild(dayLabel);

    for (let week = 0; week < 53; week++) {
        const dayOffset = week * 7 + dow - jan1Dow;
        const date = new Date(YEAR, 0, 1 + dayOffset);

        if (date.getFullYear() !== YEAR || dayOffset < 0) {
            const empty = document.createElement("div");
            empty.className = "heatmap-cell empty";
            row.appendChild(empty);
            continue;
        }

        const key = `${YEAR}-${String(date.getMonth()+1).padStart(2,"0")}-${String(date.getDate()).padStart(2,"0")}`;
        const finishCount = dateCounts[key] || 0;
        const readCount = readingDays[key] || 0;

        const cell = document.createElement("div");
        cell.className = "heatmap-cell";

        if (finishCount > 0) {
            cell.classList.add("level-finish");
            cell.title = `${key}: 완독 ${finishCount}권`;
        } else if (readCount > 0) {
            cell.classList.add("level-reading");
            cell.title = `${key}: 독서 중 (${readCount}권)`;
        } else {
            cell.classList.add("level-0");
            cell.title = key;
        }

        if (date > today) cell.classList.add("future");
        row.appendChild(cell);
    }
    container.appendChild(row);
}

const legend = document.createElement("div");
legend.className = "heatmap-legend";
legend.innerHTML = `<span>적음</span>
<div class="heatmap-cell level-0"></div>
<div class="heatmap-cell level-reading"></div>
<div class="heatmap-cell level-finish"></div>
<span>완독</span>`;
container.appendChild(legend);

dv.container.appendChild(container);
